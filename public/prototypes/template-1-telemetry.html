// Mini DB Storage Engine: Disk-Backed B+Tree
template <typename KeyType, typename ValueType>
class BPlusTreeIndex {
public:
  auto search(const KeyType& key) -> std::optional<ValueType> {
    PageGuard guard = buffer_pool_->fetch_page(root_id_);
    auto* node = reinterpret_cast<BPlusNode*>(guard.data());
    
    while (!node->is_leaf()) {
      page_id_t next_page = node->lookup_internal(key);
      guard = buffer_pool_->fetch_page(next_page); // O(log N)
      node = reinterpret_cast<BPlusNode*>(guard.data());
    }
    return node->find_in_leaf(key);
  }
};
